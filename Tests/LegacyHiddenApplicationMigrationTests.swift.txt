var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}

// Write the old wire format, including fields the migration only preserves for retry.
func archive(pid: Int32 = 123, bundle: String = "same.app", launch: Date = Date(timeIntervalSince1970: 100),
             phase: String = "hiddenOwned", boot: String = "test-boot") -> Data {
    try! JSONSerialization.data(withJSONObject: ["bootSessionID": boot, "applications": [[
        "pid": pid, "bundleID": bundle, "launchDate": launch.timeIntervalSinceReferenceDate,
        "windowIDs": [42], "sessionIDs": ["old-session"], "phase": phase
    ]]])
}
let key = LegacyHiddenApplicationMigration.defaultsKey
for phase in ["hideRequested", "hiddenOwned", "revealRequested"] {
    let defaults = TestHiddenDefaults()
    defaults.set(archive(phase: phase), forKey: key)
    NSRunningApplication.bundles = [123: "same.app", 456: "user.hidden"]
    NSRunningApplication.hidden = [123, 456]
    AccessibilityRuntimeGuard.shared.trusted = false
    identityApplications = [:] // Migration must not require readable AX windows.
    platformEvents = []
    LegacyHiddenApplicationMigration.restore(defaults: defaults, bootSessionID: "test-boot", reason: "test")
    expect(!NSRunningApplication.hidden.contains(123) && NSRunningApplication.hidden.contains(456)
           && platformEvents == ["unhide:123"], "migration restores only the identity-owned app without AX: \(phase)")
    expect(defaults.data(forKey: key) == nil, "successful migration deletes old persisted ownership: \(phase)")
    NSRunningApplication.hidden.insert(123)
    platformEvents = []
    LegacyHiddenApplicationMigration.restore(defaults: defaults, bootSessionID: "test-boot", reason: "test-repeat")
    expect(NSRunningApplication.hidden.contains(123) && platformEvents.isEmpty,
           "migration is one-time and never overrides a later user hide: \(phase)")
}
for invalid in ["pid", "bundle", "launch", "boot", "phase", "corrupt"] {
    let defaults = TestHiddenDefaults()
    NSRunningApplication.bundles = [123: "same.app"]
    NSRunningApplication.knownAppsOnly = true
    NSRunningApplication.hidden = [123]
    let data: Data
    switch invalid {
    case "pid": data = archive(pid: 999)
    case "bundle": data = archive(bundle: "other.app")
    case "launch": data = archive(launch: Date(timeIntervalSince1970: 200))
    case "boot": data = archive(boot: "old-boot")
    case "phase": data = archive(phase: "released")
    default: data = Data("invalid".utf8)
    }
    defaults.set(data, forKey: key)
    platformEvents = []
    LegacyHiddenApplicationMigration.restore(defaults: defaults, bootSessionID: "test-boot", reason: "test-invalid")
    expect(NSRunningApplication.hidden.contains(123) && platformEvents.isEmpty && defaults.data(forKey: key) == nil,
           "invalid or obsolete ownership is deleted without touching a user-hidden app: \(invalid)")
}
let defaults = TestHiddenDefaults()
defaults.set(archive(), forKey: key)
NSRunningApplication.unhideSucceeds = false
LegacyHiddenApplicationMigration.restore(defaults: defaults, bootSessionID: "test-boot", reason: "test-failure")
expect(defaults.data(forKey: key) != nil && NSRunningApplication.hidden.contains(123),
       "unconfirmed migration preserves only the recovery data needed to retry")
NSRunningApplication.onUnhide = { pid in NSRunningApplication.hidden.remove(pid) }
LegacyHiddenApplicationMigration.restore(defaults: defaults, bootSessionID: "test-boot", reason: "test-retry")
expect(defaults.data(forKey: key) == nil && !NSRunningApplication.hidden.contains(123),
       "actual visibility confirms migration even when unhide request returns false")
NSRunningApplication.onUnhide = nil
NSRunningApplication.unhideSucceeds = true

defaults.set(archive(phase: "hideRequested"), forKey: key)
NSRunningApplication.hidden = []
let lateHide = Foundation.Timer.scheduledTimer(withTimeInterval: 0.02, repeats: false) { _ in
    NSRunningApplication.hidden.insert(123)
}
LegacyHiddenApplicationMigration.restore(defaults: defaults, bootSessionID: "test-boot", reason: "test-late-hide")
expect(!NSRunningApplication.hidden.contains(123) && defaults.data(forKey: key) == nil,
       "migration settles a legacy pending hide before consuming ownership")
lateHide.invalidate()
