var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}

func resetVisibilityBackend() {
    NSRunningApplication.knownAppsOnly = false
    NSRunningApplication.hidden = []
    NSRunningApplication.terminated = []
    NSRunningApplication.hideSucceeds = true
    NSRunningApplication.unhideSucceeds = true
    NSRunningApplication.hideIsImmediate = true
    NSRunningApplication.pendingHides = []
    NSRunningApplication.bundles = [123: "same.app", 456: "other.app"]
    NSWorkspace.shared.frontmostApplication = nil
    Session.focusedElement = nil
    Session.mainElement = nil
    Session.targetedActivationByPID.removeAll()
    Session.programmaticPIDs.removeAll()
    Session.temporaryFocusReturnActivationUntilByPID.removeAll()
    TemporaryStashWindowSpaces.available = true
    TemporaryStashWindowSpaces.active = [1]
    TemporaryStashWindowSpaces.memberships = [:]
    NSScreen.screens = [NSScreen(frame: CGRect(x: 0, y: 0, width: 1000, height: 1000))]
    WindowAlphaManager.shared.windowID = nil
    applicationElements.removeAll()
    cgWindows = []
    appVisibilityEvents = []
}

func collapsedSessions(visibility: TemporaryStashAppVisibility, count: Int = 1, pid: pid_t = 123) -> [Session] {
    let app = Element()
    let windows = (0..<count).map { index -> Element in
        let window = Element()
        window.systemWindowID = UInt32(pid * 100 + Int32(index))
        return window
    }
    app.windows = windows
    applicationElements[pid] = app
    return windows.enumerated().map { index, window in
        let session = Session(appElement: app, windowElement: window, pid: pid,
                              bundleID: pid == 123 ? "same.app" : "other.app",
                              temporaryStashID: "\(pid)-\(index)")
        session.temporaryAppVisibility = visibility
        _ = session.startTemporaryShortcutStash()
        cgWindows.append([kCGWindowNumber as String: window.systemWindowID!])
        return session
    }
}

    resetVisibilityBackend()

    let lifecycleSession = collapsedSessions(visibility: lifecycle)[0]

    appVisibilityEvents.removeAll()

    case "cancel": lifecycleSession.cancelTemporaryShortcutStash()

    default: lifecycleSession.suspendForAccessibilityLoss()

    expect(!NSRunningApplication.hidden.contains(123) && appVisibilityEvents.last == "unhide:123",
           "\(action) restores SideBar-hidden app after restoring window visibility/geometry")

        expect(lifecycleSession.windowElement.frame == CGRect(x: 100, y: 200, width: 400, height: 300),
               "\(action) restores original size and position before showing app")
    }
