final class WindowSession {
    static var ignoredActivationPIDs: Set<pid_t> = []
    static func shouldIgnoreTemporaryAppActivation(activatedPID: pid_t) -> Bool {
        ignoredActivationPIDs.contains(activatedPID)
    }
    static func suppressProgrammaticActivation(for pid: pid_t, duration: TimeInterval) { ignoredActivationPIDs.insert(pid) }
    static func noteGlobalKeyDown(_ event: NSEvent) {}
    let windowElement = Element()
    func isTemporaryPinnedForFusion() -> Bool { isPinned }
    func focusForTemporaryStashTab() { focusCount += 1 }
    var focusCount = 0
    func toggleExpandCollapse() {}
    var spaceID: UInt64?
    var activeSpaceID: UInt64?
    var temporaryStashSlot = 0
    let temporaryStashID: String
    var isManagedByTemporaryShortcut = true
    var isExpandedForCoordination = false
    var isPinned = false
    var edge = 1
    var screenID: UInt32 = 1
    var visibleFrame = mainScreen
    var pid: pid_t = 123
    var activationFocused = false
    var programmaticActivation = false
    var targetedActivation = false
    var activationSuppressionUntil = Date.distantPast
    var eligible = true
    var canHide = true
    var canReveal = true
    var hoverLock = false
    var appActivationReveal = false
    init(_ id: String, edge: Int = 1) {
        temporaryStashID = id; self.edge = edge
    }
    func temporaryStashTabDescriptor() -> TemporaryStashTabDescriptor? {
        return TemporaryStashTabDescriptor(session: self, edge: edge, screenID: screenID,
                                           visibleFrame: visibleFrame,
                                           isOnScreen: eligible, title: temporaryStashID, spaceID: spaceID, activeSpaceID: activeSpaceID)
    }
    func temporaryActivationDiagnostics(activatedPID: pid_t) -> String { temporaryStashID }
    func cancelTemporaryActivationReveal() {}
    var focusOutside = false
    func temporaryActivationFocusIsOutside(_ managed: [WindowSession]) -> Bool {
        focusOutside || !isManagedByTemporaryShortcut || !managed.contains { $0.pid == pid && !$0.isExpandedForCoordination } || programmaticActivation || targetedActivation || Date() < activationSuppressionUntil
    }
    func canRevealForTemporaryAppActivation(activatedPID: pid_t, requiresFrontmost: Bool = true) -> Bool {
        isManagedByTemporaryShortcut && !isExpandedForCoordination && activationFocused
            && !programmaticActivation && !targetedActivation && Date() >= activationSuppressionUntil
            && activatedPID == pid && (!requiresFrontmost || NSWorkspace.shared.frontmostApplication?.processIdentifier == pid)
    }
    func setFusionHoverLock(_ locked: Bool) { hoverLock = locked }
    func hideForTemporaryAppActivation(activatedPID: pid_t) -> Bool {
        guard isManagedByTemporaryShortcut, isExpandedForCoordination, pid != activatedPID, !isPinned else { return false }
        return hideForTemporaryStashTab()
    }
    func hideForTemporaryStashTab() -> Bool {
        if isPinned { return true }
        guard canHide else { return false }
        transitions.append("hide:" + temporaryStashID)
        isExpandedForCoordination = false
        return true
    }
    func beginTemporaryStashReveal() -> Bool { true }
    func revealForTemporaryStashTab(isAppActivated: Bool = false) {
        guard canReveal else { return }
        appActivationReveal = isAppActivated
        transitions.append("reveal:" + temporaryStashID)
        isExpandedForCoordination = true
    }
}

final class Timer {
    static var all: [Timer] = []
    let due: Double
    let action: (Timer) -> Void
    var valid = true
    init(timeInterval: Double, repeats: Bool, block: @escaping (Timer) -> Void) {
        due = Clock.now + timeInterval
        action = block
        Self.all.append(self)
    }
    func invalidate() { valid = false }
    static func fireDue() {
        let due = all.filter { $0.valid && $0.due <= Clock.now + 1e-9 }
        for timer in due { timer.valid = false; timer.action(timer) }
    }
}

final class TemporaryStashTabWindow {
    static var live: [String: CGRect] = [:]
    static var instances: [String: TemporaryStashTabWindow] = [:]
    static var placeholders: [ObjectIdentifier: CGRect] = [:]
    static var created = 0
    static var closed = 0
    var title = ""
    var frame = CGRect.zero
    var onEvent: ((TemporaryStashTabEvent) -> Void)?
    init() { Self.created += 1 }
    func update(frame: CGRect, title: String, expanded: Bool, isPlaceholder: Bool = false) {
        if isPlaceholder { Self.placeholders[ObjectIdentifier(self)] = frame; return }
        self.title = title
        self.frame = frame
        Self.live[title] = frame
        Self.instances[title] = self
    }
    func orderFrontRegardless() { Self.live[title] = frame }
    func orderOut(_ sender: Any?) { Self.live[title] = nil }
    func close() { Self.placeholders[ObjectIdentifier(self)] = nil; Self.closed += 1; Self.live[title] = nil; Self.instances[title] = nil }
    static func emit(_ title: String, _ event: TemporaryStashTabEvent) {
        switch event {
        case .down: NSEvent.pressedMouseButtons = 1
        case .up: NSEvent.pressedMouseButtons = 0
        default: break
        }
        instances[title]?.onEvent?(event)
    }
}
