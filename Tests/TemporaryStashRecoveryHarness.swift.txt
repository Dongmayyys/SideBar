// The production archive and manager entry points run with in-memory AX and defaults.
final class UserDefaults {
    static let standard = UserDefaults()
    var values: [String: Data] = [:]
    var writes = 0
    func data(forKey key: String) -> Data? { values[key] }
    func set(_ data: Data, forKey key: String) { values[key] = data; writes += 1 }
}
var applicationElements: [Int32: Element] = [:]
func testApplicationElement(_ pid: Int32) -> Element { applicationElements[pid] ?? Element() }
final class AppConfig {
    static let shared = AppConfig()
    var enabled: Set<String> = []
    func isAppEnabled(bundleID: String) -> Bool { enabled.contains(bundleID) }
}
final class TemporaryStashTabCoordinator {
    var pendingActivationPID: pid_t? { nil }
    var assignments: [TemporaryStashSlotAssignment] = []
    var suppressedIDs: Set<String> = []
    func restoreSlots(_ assignments: [TemporaryStashSlotAssignment], pendingIDs: Set<String> = []) { self.assignments = assignments }
    func suppressRestoredHover(for session: Session) { suppressedIDs.insert(session.temporaryStashID) }
}
final class RecoveryManager {
    var sessions: [Session] = []
    let temporaryTabCoordinator = TemporaryStashTabCoordinator()
    let temporaryAppVisibility = TemporaryStashAppVisibility()
    var temporaryOrdinals = TemporaryStashOrdinals()
    let temporaryRecoveryStore: TemporaryStashRecoveryStore
    var temporaryRecovery: TemporaryStashRecoverySnapshot
    var pendingTemporaryRecovery: [TemporaryStashRecoveryRecord]
    var temporaryRecoveryRetry: DispatchWorkItem?
    var hasCleanedUp = false
    var temporarilyExcludedWindowKeys: Set<String> = []
    var dockExclusions: Set<String> = []
    init(_ records: [TemporaryStashRecoveryRecord], defaults: UserDefaults = UserDefaults()) {
        temporaryRecoveryStore = TemporaryStashRecoveryStore(defaults: defaults, bootSessionID: "test-boot")
        temporaryRecovery = TemporaryStashRecoverySnapshot()
        temporaryRecovery.update(windows: records)
        pendingTemporaryRecovery = temporaryRecovery.windows
        for record in records {
            temporaryOrdinals.reserve(bundleID: record.bundleID, windowID: record.id, ordinal: record.ordinal)
        }
        temporaryTabCoordinator.restoreSlots(temporaryRecovery.assignments)
    }
    func attachSessionLifecycle(to session: Session) {
        session.temporaryAppVisibility = temporaryAppVisibility
        session.onTemporaryShortcutSessionEnded = { [weak self] session, exclude, remove in
            self?.handleTemporaryShortcutSessionEnded(session, excludeWindow: exclude, removeSession: remove)
        }
    }
    func reconcileIndicators() {
        persistTemporaryRecovery()
        temporaryAppVisibility.reconcile(sessions: sessions)
    }
    func monitorApp(pid: pid_t, bundleID: String) {}
    var focusedTarget: (appElement: Element, windowElement: Element, pid: pid_t, bundleID: String)?
    func currentFocusedStandardWindow() -> (appElement: Element, windowElement: Element, pid: pid_t, bundleID: String)? { focusedTarget }
    func clearTemporarilyExcludedWindow(for window: Element, pid: pid_t) {}
    func stash(_ window: Element, pid: pid_t = 123, bundleID: String = "same.app") -> Bool {
        focusedTarget = (testApplicationElement(pid), window, pid, bundleID)
        return handleTemporaryShortcut()
    }
    func terminated(pid: pid_t) { retainTemporaryRecoveryAfterTermination(pid: pid) }
    func refreshSideBarHotkeyRuntime() {}
    func windowKey(for window: Element, pid: Int32) -> String? { "\(pid):\(window.systemWindowID ?? 42)" }
    func refreshTemporaryDockExclusion(bundleID: String) {
        if temporaryOrdinals.hasMembers(bundleID: bundleID) { dockExclusions.insert(bundleID) }
        else { dockExclusions.remove(bundleID) }
    }
    func run(attemptsRemaining: Int = 0) {
        temporaryRecoveryRetry?.cancel()
        restoreTemporaryShortcuts(attemptsRemaining: attemptsRemaining)
    }
    func persist() { persistTemporaryRecovery() }
    func resolveCloses(now: Date) { resolveClosedTemporaryRecovery(now: now); persistTemporaryRecovery() }
    func stop() {
        stopTemporaryRecovery()
        sessions.forEach { $0.restoreAndDestroy() }
        temporaryAppVisibility.restoreAll(reason: "cleanup")
        sessions.removeAll()
    }

__MANAGER_METHODS__
}
