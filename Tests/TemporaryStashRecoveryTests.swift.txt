var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}
let primary = CGRect(x: 0, y: 0, width: 1000, height: 1000)
let original = CGRect(x: 130, y: 200, width: 400, height: 300)
func record(_ id: String = "saved", pid: Int32 = 123, bundle: String = "same.app", windowID: UInt32 = 42,
            edge: Int = 2, screenID: UInt32 = 1, ordinal: Int = 2) -> TemporaryStashRecoveryRecord {
    TemporaryStashRecoveryRecord(id: id, bundleID: bundle, pid: pid, windowID: windowID,
                                 edge: edge, screenID: screenID, originalFrame: original, ordinal: ordinal)
}
func resetBackend() {
    NSScreen.screens = [NSScreen(frame: primary)]
    NSRunningApplication.knownAppsOnly = true
    NSRunningApplication.bundles = [123: "same.app"]
    NSRunningApplication.terminated = []
    AppConfig.shared.enabled = []
    WindowAlphaManager.shared.windowID = nil
    let app = Element()
    let window = Element()
    window.systemWindowID = 42
    app.windows = [window]
    applicationElements = [123: app]
}
let saved = record()
expect(saved.matches(pid: 123, bundleID: "same.app", windowID: 42), "identity requires all three fields")
expect(!saved.matches(pid: 124, bundleID: "same.app", windowID: 42), "different pid rejected")
expect(!saved.matches(pid: 123, bundleID: "other.app", windowID: 42), "different bundle rejected")
expect(!saved.matches(pid: 123, bundleID: "same.app", windowID: 43), "different system window rejected")
var snapshot = TemporaryStashRecoverySnapshot()
snapshot.update(windows: [record("one", ordinal: 1), saved], liveOrders: [])
snapshot.orders = [TemporaryStashTabOrder(screenID: 1, edge: 2, ids: [saved.id, "one"])]
let data = try! JSONEncoder().encode(snapshot)
expect(try! JSONDecoder().decode(TemporaryStashRecoverySnapshot.self, from: data) == snapshot,
       "record and per-display edge order encode/decode without losing original frame or ordinal")
let defaults = UserDefaults()
let store = TemporaryStashRecoveryStore(defaults: defaults, bootSessionID: "boot-a")
store.save(snapshot)
expect(store.load() == snapshot, "archive roundtrip through isolated in-memory defaults")
store.save(snapshot)
expect(defaults.writes == 1, "unchanged polling snapshot does not rewrite defaults")
expect(TemporaryStashRecoveryStore(defaults: defaults, bootSessionID: "boot-b").load().windows.isEmpty,
       "computer reboot discards old identities even if pid and window ID are reused")
expect(TemporaryStashRecoveryStore(defaults: defaults, bootSessionID: "").load().windows.isEmpty,
       "unreadable boot identity never guesses")
defaults.values[TemporaryStashRecoveryStore.defaultsKey] = Data("broken".utf8)
expect(store.load().windows.isEmpty, "corrupt archive safely loads empty")
expect(!TemporaryStashRecoveryStore.currentBootSessionID().isEmpty, "macOS supplies stable boot session UUID")

var slots = TemporaryStashRecoverySnapshot()
let a = record("a", ordinal: 1), b = record("b", ordinal: 2), c = record("c", ordinal: 3)
slots.update(windows: [a, b, c], liveOrders: [])
slots.update(windows: [a, b, c], liveOrders: [TemporaryStashTabOrder(screenID: 1, edge: 2, ids: ["c", "a"])])
expect(slots.orders[0].ids == ["c", "b", "a"], "sorting live tabs retains pending AX window's slot")
slots.remove(id: "b")
expect(slots.windows.map(\.id) == ["a", "c"] && slots.orders[0].ids == ["c", "a"],
       "cancel or close removes record and only its order slot")
let moved = record("a", edge: 1, screenID: 2, ordinal: 1)
slots.update(windows: [moved, c], liveOrders: [])
expect(slots.orders.contains { $0.screenID == 2 && $0.edge == 1 && $0.ids == ["a"] }
       && slots.orders.first { $0.screenID == 1 }?.ids == ["c"], "edge and screen changes migrate order membership")

var ordinals = TemporaryStashOrdinals()
ordinals.reserve(bundleID: "same.app", windowID: "late", ordinal: 8)
ordinals.reserve(bundleID: "same.app", windowID: "early", ordinal: 2)
expect(ordinals.assign(bundleID: "same.app", windowID: "early") == 2, "out-of-order restoration keeps saved ordinal")
expect(ordinals.assign(bundleID: "same.app", windowID: "new") == 9, "new stash starts beyond all restored and pending ordinals")
ordinals.release(bundleID: "same.app", windowID: "late")
expect(ordinals.assign(bundleID: "same.app", windowID: "next") == 10, "removed restored window does not reuse its suffix")

resetBackend()
let manager = RecoveryManager([saved])
manager.run()
expect(manager.sessions.count == 1 && manager.pendingTemporaryRecovery.isEmpty, "startup restores existing exact window once")
let restored = manager.sessions[0]
expect(restored.isSnappedForCoordination && restored.alpha == 0 && !restored.isTemporarilyPinned,
       "restored window is collapsed, invisible and unpinned")
expect(restored.currentEdge == 2 && restored.snappedScreen?.displayID == 1, "saved edge beats nearest edge on saved display")
expect(restored.temporaryOriginalFrame == original && restored.temporaryStashOrdinal == 2,
       "restoration injects earliest original frame and unchanged ordinal")
expect(!restored.focusReleasePending && NSWorkspace.shared.frontmostApplication == nil,
       "restore neither activates a running application nor schedules focus release")
expect(manager.temporaryTabCoordinator.suppressedIDs == [saved.id], "stationary pointer cannot auto-reveal restored tab")
expect(manager.temporaryOrdinals.assign(bundleID: "same.app", windowID: "new") == 3,
       "manager reserves saved ordinal before accepting any new stash")
expect(manager.temporaryRecoveryStore.load().windows == [saved], "successful restoration persists original identity and geometry")
restored.cancelTemporaryShortcutStash()
expect(restored.windowElement.frame.origin == original.origin && restored.alpha == 1,
       "cancel restored window returns to earliest original, not rescue/re-stash position")
expect(manager.temporaryRecoveryStore.load().windows.isEmpty, "user cancellation immediately removes archive entry")

resetBackend()
let shutdown = RecoveryManager([saved])
shutdown.run()
shutdown.sessions[0].isTemporarilyPinned = true
shutdown.stop()
expect(shutdown.temporaryRecoveryStore.load().windows == [saved], "normal cleanup preserves intent despite session-ended callback")
expect(applicationElements[123]!.windows![0].frame.origin == original.origin,
       "normal cleanup still physically returns window to its original position")
let reopen = RecoveryManager(shutdown.temporaryRecoveryStore.load().windows)
reopen.run()
expect(reopen.sessions.count == 1 && !reopen.sessions[0].isTemporarilyPinned, "manual exit and reopen resets pin state")
reopen.sessions[0].closed()
expect(reopen.temporaryRecoveryStore.load().windows.isEmpty, "window close immediately deletes persisted recovery")

resetBackend()
let newStash = RecoveryManager([])
let fresh = Session(windowElement: applicationElements[123]!.windows![0], temporaryStashID: "fresh")
expect(fresh.startTemporaryShortcutStash(), "normal shortcut entry still stashes")
fresh.temporaryStashOrdinal = newStash.temporaryOrdinals.assign(bundleID: fresh.bundleID, windowID: fresh.temporaryStashID)
newStash.attachSessionLifecycle(to: fresh)
newStash.sessions.append(fresh)
newStash.persist()
expect(newStash.temporaryRecoveryStore.load().windows.first?.originalFrame == original.offsetBy(dx: -30, dy: 0),
       "normal stash records pre-stash geometry")
fresh.currentEdge = 2
newStash.persist()
expect(newStash.temporaryRecoveryStore.load().windows.first?.edge == 2, "live edge change updates archive")
fresh.cancelTemporaryShortcutStash(keepingFrame: CGRect(x: 350, y: 400, width: 400, height: 300))
expect(newStash.temporaryRecoveryStore.load().windows.isEmpty, "drag-away cancellation deletes recovery too")

for mismatch in ["absent", "pid", "bundle", "window", "unknownID", "nonstandard", "configured", "terminated"] {
    resetBackend()
    switch mismatch {
    case "absent": applicationElements[123]!.windows = []
    case "pid": NSRunningApplication.bundles = [999: "same.app"]
    case "bundle": NSRunningApplication.bundles[123] = "other.app"
    case "window": applicationElements[123]!.windows![0].systemWindowID = 999
    case "unknownID": applicationElements[123]!.windows![0].systemWindowID = nil
    case "nonstandard": applicationElements[123]!.windows![0].standard = false
    case "configured": AppConfig.shared.enabled.insert("same.app")
    case "terminated": NSRunningApplication.terminated.insert(123)
    default: break
    }
    let candidate = RecoveryManager([saved])
    candidate.run()
    expect(candidate.sessions.isEmpty && candidate.pendingTemporaryRecovery.isEmpty
           && candidate.temporaryRecoveryStore.load().windows.isEmpty, "invalid startup candidate discarded: \(mismatch)")
}
resetBackend()
applicationElements[123]!.windows = nil
let delayed = RecoveryManager([saved])
delayed.run(attemptsRemaining: 1)
expect(delayed.sessions.isEmpty && delayed.pendingTemporaryRecovery == [saved]
       && delayed.temporaryRecoveryStore.load().windows == [saved], "transient AX enumeration failure retains record for bounded retry")
expect(delayed.temporaryOrdinals.assign(bundleID: "same.app", windowID: "while-waiting") == 3,
       "new stash during AX retry cannot collide with pending ordinal")
let lateWindow = Element(); lateWindow.systemWindowID = 42
applicationElements[123]!.windows = [lateWindow]
delayed.run()
expect(delayed.sessions.count == 1 && delayed.pendingTemporaryRecovery.isEmpty, "later AX success restores pending window")
resetBackend()
applicationElements[123]!.windows![0].frameUnavailable = true
// An unreadable frame is a transient condition; final exhaustion must remove its record.
let failedAX = RecoveryManager([saved])
failedAX.run(attemptsRemaining: 1)
failedAX.run()
expect(failedAX.pendingTemporaryRecovery.isEmpty && failedAX.temporaryRecoveryStore.load().windows.isEmpty,
       "retry exhaustion discards record rather than restoring a guessed window")
resetBackend()
applicationElements[123]!.windows = nil
let interrupted = RecoveryManager([saved])
interrupted.run(attemptsRemaining: 1)
let scheduled = interrupted.temporaryRecoveryRetry
interrupted.stop()
expect(scheduled?.isCancelled == true && interrupted.temporaryRecoveryStore.load().windows == [saved],
       "cleanup cancels queued startup retry and retains pending record")

resetBackend()
let external = NSScreen(frame: CGRect(x: 1200, y: 1000, width: 1000, height: 1000), displayID: 2)
NSScreen.screens.append(external)
let externalManager = RecoveryManager([record(screenID: 2)])
externalManager.run()
expect(externalManager.sessions.first?.snappedScreen?.displayID == 2
       && externalManager.sessions.first?.windowElement.frame.minX == 2199,
       "saved physical display is preferred even when visible window now occupies another display")
expect(externalManager.sessions.first!.windowElement.frame.minY >= external.frame.minY,
       "restoring to vertically displaced display clamps window into its bounds")
externalManager.sessions[0].cancelTemporaryShortcutStash()
expect(externalManager.sessions[0].windowElement.frame.origin == original.origin, "cross-display restore still cancels to earliest original")
resetBackend()
let unplugged = RecoveryManager([record(screenID: 999)])
unplugged.run()
expect(unplugged.sessions.first?.currentEdge == 1 && unplugged.sessions.first?.snappedScreen?.displayID == 1,
       "unplugged display falls back to current screen and existing nearest-edge rule")
resetBackend()
NSScreen.screens.append(NSScreen(frame: primary.offsetBy(dx: 1000, dy: 0), displayID: 2))
let blockedRight = RecoveryManager([saved])
blockedRight.run()
expect(blockedRight.sessions.first?.currentEdge == 1, "saved edge newly blocked by neighbor selects available opposite edge")
expect(blockedRight.temporaryRecoveryStore.load().windows.first?.edge == 1, "fallback direction is written back for next reopen")
NSScreen.screens.append(NSScreen(frame: primary.offsetBy(dx: -1000, dy: 0), displayID: 3))
let bothBlocked = RecoveryManager([saved])
bothBlocked.run()
expect(bothBlocked.sessions.isEmpty && bothBlocked.temporaryRecoveryStore.load().windows.isEmpty,
       "two blocked sides abandon this window and delete its record")

print("RESULT checks=\(total) failures=\(failures)")
exit(failures == 0 ? 0 : 1)
