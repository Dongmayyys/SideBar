}

@discardableResult
func axSetPositionAttribute(on element: Element, position: CGPoint, context: String) -> AXResult {
    element.writes += 1
    element.events.append("position")
    let result = element.positionResults.isEmpty ? .success : element.positionResults.removeFirst()
    if result == .success { element.frame.origin = position }
    return result
}
@discardableResult
func axSetBoolAttribute(on element: Element, attribute: CFString, value: Bool, context: String) -> AXResult { .success }

        let screen = detectDisplayBounds()
        let x = currentEdge == 1 ? screen.minX : screen.maxX - frame.width
        let hidden = currentEdge == 1 ? screen.minX - frame.width + 1 : screen.maxX - 1
        return (CGPoint(x: x, y: frame.minY), .zero, CGPoint(x: hidden, y: frame.minY))
    }
    // Collapse's real control flow has its own production extraction suite.
    func collapseWindow(immediate: Bool, triggerEffects: Bool) -> Bool {
        guard !isAnimating, let frame = getRawWindowFrame(), let geometry = edgeGeometry(for: frame) else { return false }
        state = .snapped(edge: currentEdge, hiddenOrX: geometry.visualHidden.x)
        windowElement.frame.origin = geometry.visualHidden
        alpha = 0
        return true
    }
