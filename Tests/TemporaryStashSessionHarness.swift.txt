enum AccessibilityRuntimeGuard {
    static let shared = RuntimeTrust()
    static func copyAttributeValue(of element: Element, attribute: CFString, value: inout CFTypeRef?, context: String) -> AXError {
        if attribute as String == "AXRole" {
            value = element.role as CFTypeRef?
            return value == nil ? .noValue : .success
        }
        if attribute as String == "AXWindows" {
            value = element.windows as CFTypeRef?
            return element.windows == nil ? .cannotComplete : .success
        }
        if attribute as String == "AXMainWindow" || attribute as String == "AXFocusedWindow" {
            value = (attribute as String == "AXMainWindow" ? element.mainWindow : element.focusedWindow) as CFTypeRef?
            return value == nil ? .noValue : .success
        }
        if attribute as String == "AXPosition" {
            var point = element.frame.origin
            value = AXValueCreate(.cgPoint, &point)
            return element.frameUnavailable ? .cannotComplete : .success
        }
        if attribute as String == "AXSize" {
            var size = element.frame.size
            value = AXValueCreate(.cgSize, &size)
            return element.frameUnavailable ? .cannotComplete : .success
        }
        axEvents.append("read")
        value = element.enhancedUI.map { $0 ? kCFBooleanTrue : kCFBooleanFalse }
        return element.enhancedUIReadResult
    }
}

final class NSRunningApplication {
    let processIdentifier: Int32
    static var knownAppsOnly = false
    static var bundles: [Int32: String] = [:]
    static var terminated: Set<Int32> = []
    static var hidden: Set<Int32> = []
    static var hideSucceeds = true
    static var unhideSucceeds = true
    static var unhideIsImmediate = true
    static var onUnhide: ((Int32) -> Void)?
    static var pendingUnhides: Set<Int32> = []
    static var hideIsImmediate = true
    static var pendingHides: Set<Int32> = []
    static var launches: [Int32: Date] = [:]
    var launchDate: Date? { Self.launches[processIdentifier] ?? Date(timeIntervalSince1970: 100) }
    var processStartDate: Date? { launchDate }
    var isHidden: Bool { Self.hidden.contains(processIdentifier) }
    var bundleIdentifier: String? { Self.bundles[processIdentifier] ?? "same.app" }
    var isTerminated: Bool { Self.terminated.contains(processIdentifier) }
    init?(processIdentifier: Int32) {
        if Self.knownAppsOnly && Self.bundles[processIdentifier] == nil { return nil }
        self.processIdentifier = processIdentifier
    }
    func activate(options: ActivateOptions) {
        Self.hidden.remove(processIdentifier)
        NSWorkspace.shared.frontmostApplication = self
    }
    func hide() -> Bool {
        appVisibilityEvents.append("hide:\(processIdentifier)")
        if Self.hideSucceeds {
            if Self.hideIsImmediate { Self.hidden.insert(processIdentifier) }
            else { Self.pendingHides.insert(processIdentifier) }
        }
        return Self.hideSucceeds
    }
    func unhide() -> Bool {
        appVisibilityEvents.append("unhide:\(processIdentifier)")
        if Self.unhideSucceeds {
            if Self.unhideIsImmediate {
                Self.hidden.remove(processIdentifier)
                Self.pendingHides.remove(processIdentifier)
            } else { Self.pendingUnhides.insert(processIdentifier) }
        }
        Self.onUnhide?(processIdentifier)
        return Self.unhideSucceeds
    }
}

@discardableResult
func axSetPositionAttribute(on element: Element, position: CGPoint, context: String) -> AXResult {
    appVisibilityEvents.append("position")
    element.writes += 1
    element.events.append("position")
    axEvents.append("position")
    let result = element.positionResults.isEmpty ? .success : element.positionResults.removeFirst()
    if result == .success { element.frame.origin = position }
    return result
}
@discardableResult
func axSetSizeAttribute(on element: Element, size: CGSize, context: String) -> AXResult {
    appVisibilityEvents.append("size")
    element.events.append("size")
    axEvents.append("size")
    let result = element.sizeResults.isEmpty ? .success : element.sizeResults.removeFirst()
    if result == .success {
        element.frame.size = CGSize(width: size.width, height: min(size.height, element.maximumHeight ?? size.height))
    }
    return result
}
@discardableResult
func axSetBoolAttribute(on element: Element, attribute: CFString, value: Bool, context: String) -> AXResult {
    guard attribute as String == "AXEnhancedUserInterface" else {
        element.boolAttributes[attribute as String] = value
        return .success
    }
    axEvents.append(value ? "restore" : "disable")
    let result = element.boolResults.isEmpty ? .success : element.boolResults.removeFirst()
    if result == .success { element.enhancedUI = value }
    return result
}
