var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}

    expect(s.startTemporaryShortcutStash() && s.isSnappedForCoordination && s.alpha == 0,
           "production start hides left/right positioned window at right edge")

           "production reveal fills usable height on managed edge and starts tracking")

           "cancel clears pin and pending focus/animation work")

           "successful restoration clears rescue record and stops tracking")

NSScreen.screens.append(NSScreen(frame: CGRect(x: -1000, y: 0, width: 1000, height: 1000), displayID: 2))
expect(!screenStash.snappedScreenIsStale(in: NSScreen.screens) && screenStash.currentEdge == 1,
       "attaching adjacent display does not reroute an existing stash")
expect(screenStash.reseat(in: NSScreen.screens) && screenStash.currentEdge == 1,
       "same physical display never selects a new edge, even when blocked")
NSScreen.screens.removeLast()
expect(!screenStash.snappedScreenIsStale(in: NSScreen.screens) && screenStash.currentEdge == 1,
       "unplugging a neighbor does not move an existing edge")
let oldSlot = screenStash.temporaryStashSlot
NSScreen.screens = [NSScreen(frame: CGRect(x: 0, y: 0, width: 1200, height: 1000), displayID: 9)]
expect(screenStash.snappedScreenIsStale(in: NSScreen.screens)
       && screenStash.reseat(in: NSScreen.screens)
       && screenStash.currentEdge == 2 && screenStash.snappedScreen?.displayID == 9
       && screenStash.temporaryStashSlot == oldSlot,
       "losing original display selects right on new display and retains preferred slot")
expect(screenStash.temporaryOriginalFrame == screenOriginal && screenStash.alpha == 0 && !screenStash.focusReleasePending,
       "screen fallback preserves original cancel position without focus release")
resetScreens()

       "new production stash uses left only when right is blocked")
