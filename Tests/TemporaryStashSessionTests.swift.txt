var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}
func resetScreens() { NSScreen.screens = [NSScreen(frame: CGRect(x: 0, y: 0, width: 1000, height: 1000))] }

for right in [false, true] {
    resetScreens()
    let s = Session()
    s.windowElement.frame.origin.x = right ? 550 : 100
    let original = s.windowElement.frame
    var excluded: Bool?
    s.onTemporaryShortcutSessionEnded = { _, exclude, _ in excluded = exclude }
    expect(s.startTemporaryShortcutStash() && s.isSnappedForCoordination && s.alpha == 0,
           "production start hides window edge=\(right ? 2 : 1)")
    expect(s.temporaryOriginalFrame == original && s.currentEdge == (right ? 2 : 1), "start stores original and nearest edge")
    s.revealForTemporaryStashTab()
    expect(s.isExpandedForCoordination && s.alpha == 1 && s.windowElement.frame.minY == original.minY && s.tracking,
           "production reveal preserves vertical position and starts tracking")
    s.isTemporarilyPinned = true
    expect(s.hideForTemporaryStashTab() && s.isExpandedForCoordination, "production switch leaves pinned window visible")
    s.windowElement.events.removeAll()
    s.cancelTemporaryShortcutStash()
    expect(s.windowElement.events == ["position", "alpha"], "cancel restores position before alpha")
    expect(s.windowElement.frame.origin == original.origin && s.alpha == 1 && s.destroyed && excluded == true,
           "production pinned cancel restores original and excludes only this window")
    expect(!s.isMirrorPinActive && !s.isTemporarilyPinned && !s.focusReleasePending && s.windowAnimator.stopped,
           "cancel clears pin and pending focus/animation work")
    expect(s.clearedRecovery && !s.screenLocked && s.mouseTrackingTimer?.invalidated == true,
           "successful restoration clears rescue record and stops tracking")
}

for pinned in [false, true] {
    resetScreens()
    let drag = Session()
    var dragExcluded = false
    drag.onTemporaryShortcutSessionEnded = { _, excluded, _ in dragExcluded = excluded }
    _ = drag.startTemporaryShortcutStash()
    drag.revealForTemporaryStashTab()
    drag.isTemporarilyPinned = pinned
    drag.windowElement.frame.origin.x = 250
    let dropped = drag.windowElement.frame
    drag.windowElement.writes = 0
    drag.isDragging = true
    drag.handleMouseUp()
    expect(drag.destroyed && drag.windowElement.frame == dropped && !drag.isTemporaryShortcutManaged,
           "production drag detach cancels at dropped frame pinned=\(pinned)")
    expect(drag.windowElement.writes == 0 && drag.alpha == 1 && dragExcluded && drag.clearedRecovery,
           "drag cancel restores visibility and exclusion without position writes pinned=\(pinned)")
}

let focusedA = Session()
let focusedB = Session()
_ = focusedA.startTemporaryShortcutStash()
_ = focusedB.startTemporaryShortcutStash()

Session.programmaticPIDs.removeAll()
NSWorkspace.shared.frontmostApplication = NSRunningApplication(processIdentifier: 123)

expect(focusedA.canRevealForTemporaryAppActivation(activatedPID: 123)
       && !focusedB.canRevealForTemporaryAppActivation(activatedPID: 123),
       "production activation targets only focused stash even when another stash is main")
Session.focusedElement = nil
expect(focusedB.canRevealForTemporaryAppActivation(activatedPID: 123)
       && !focusedA.canRevealForTemporaryAppActivation(activatedPID: 123),
       "production activation falls back to main only if focused is absent")
Session.focusedElement = Element()
expect(!focusedA.canRevealForTemporaryAppActivation(activatedPID: 123)
       && !focusedB.canRevealForTemporaryAppActivation(activatedPID: 123),
       "production non-stashed focus prevents main-window fallback")
Session.focusedElement = focusedB.windowElement
focusedA.revealForTemporaryStashTab(isAppActivated: true)
expect(focusedA.pendingDockInteraction && focusedA.hasReleasedDockClick,
       "keyboard application activation waits for interaction even with pointer outside")
expect(!focusedB.canRevealForTemporaryAppActivation(activatedPID: 123),
       "production targeted reveal cannot reenter on still-focused sibling")
expect(TemporaryStashShortcutAction.resolve(isConfiguredApp: false, isManaged: true,
                                          isExpanded: focusedA.isExpandedForCoordination) == .cancel,
       "revealed window can immediately cancel via stash shortcut")
Session.targetedActivationByPID.removeAll()
Session.programmaticPIDs.insert(123)
expect(!focusedB.canRevealForTemporaryAppActivation(activatedPID: 123), "production focus-return activation is ignored")
Session.programmaticPIDs.removeAll()
expect(!focusedB.canRevealForTemporaryAppActivation(activatedPID: 999), "activation for another app is ignored")
NSWorkspace.shared.frontmostApplication = NSRunningApplication(processIdentifier: 999)

NSWorkspace.shared.frontmostApplication = NSRunningApplication(processIdentifier: 123)

focusedB.activationSuppressionUntil = .distantPast

focusedB.isTemporaryShortcutManaged = false

    resetScreens()
    let fast = Session()
    fast.currentEdge = edge
    var callbacks: [String] = []
    fast.runPositionHelper(onStart: { callbacks.append("start") }, completion: { callbacks.append("finish") })
    expect(fast.windowAnimator.interpolations == 0 && fast.windowElement.frame.minY == 250 && callbacks == ["start", "finish"],
           "shared position helper skips interpolation and finishes callbacks for temporary pin release edge=\(edge)")
    fast.isTemporaryShortcutManaged = false
    fast.runPositionHelper(completion: {})
