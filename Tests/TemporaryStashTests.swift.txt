var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}
func near(_ left: CGFloat, _ right: CGFloat) -> Bool { abs(left - right) < 0.001 }

expect(isolated.allSatisfy { $0.minX == 0 && $0.width == 24 && $0.height == 120 }, "fixed label dimensions at left edge")
let pair = TemporaryStashGeometry.labelFrames(centers: [100, 500, 540, 900], edge: 2, visibleFrame: mainScreen)

expect(pair[0].midY == 100 && pair[3].midY == 900, "unrelated labels do not move")
expect(pair.allSatisfy { $0.maxX == 1000 }, "right labels align to right edge")
let chain = TemporaryStashGeometry.labelFrames(centers: [300, 320, 330], edge: 1, visibleFrame: mainScreen)
expect(near(chain[1].midY - chain[0].midY, 120) && near(chain[2].midY - chain[1].midY, 120), "chain collisions settle without overlap and preserve order")
expect(near(chain.reduce(0) { $0 + $1.midY }, 950), "collision resolution preserves cluster center")
expect(TemporaryStashGeometry.labelFrames(centers: [], edge: 1, visibleFrame: mainScreen).isEmpty, "empty edge has no labels")

// Deterministic spread of dense layouts: order, no overlap, untouched distant label.
for count in 2...12 {
    let centers = (0..<count).map { CGFloat($0 * $0 * 7 + 200) }
    let frames = TemporaryStashGeometry.labelFrames(centers: centers, edge: 1, visibleFrame: mainScreen)
    expect(zip(frames, frames.dropFirst()).allSatisfy { $0.maxY <= $1.minY + 0.001 }, "dense layout order and non-overlap count=\(count)")
}

    NSEvent.mouseLocation = point
    coordinator.reconcile(sessions: sessions)

    coordinator.reconcile(sessions: sessions)

let otherEdge = WindowSession("otherEdge", center: 500, edge: 2)

NSEvent.mouseLocation = CGPoint(x: 500, y: 400)
active1.isExpandedForCoordination = true
active2.activationFocused = true
pinned.isExpandedForCoordination = true
pinned.isPinned = true
otherEdge.isExpandedForCoordination = true
transitions.removeAll()
activationCoordinator.handleAppActivation(activatedPID: 123, sessions: activationSessions)

       "application activation reveals focused stashed window after hiding same-edge unpinned window")
expect(active2.appActivationReveal, "application activation uses existing Dock interaction wait mode")
expect(pinned.isExpandedForCoordination && otherEdge.isExpandedForCoordination,
       "application activation leaves pinned and other-edge windows expanded")

active2.isExpandedForCoordination = false
active2.activationFocused = false
transitions.removeAll()
activationCoordinator.handleAppActivation(activatedPID: 123, sessions: activationSessions)

       "activation retries failed hide using existing reconcile timer")

active1.isExpandedForCoordination = true
