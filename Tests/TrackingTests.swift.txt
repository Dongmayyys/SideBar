var failures = 0
var total = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    print("\(condition ? "PASS" : "FAIL") \(name)")
    if !condition { failures += 1 }
}
func reset(edge: Int = 1) {
    Clock.time = 0; Timer.all.removeAll(); DispatchQueue.blocks.removeAll()
    NSEvent.pressedMouseButtons = 0
    NSEvent.mouseLocation = CGPoint(x: edge == 1 ? 900 : 100, y: 700)
    NSWorkspace.shared.frontmostApplication = nil
}
for edge in [1, 2] {
    for dockClick in [false, true] {
        reset(edge: edge)
        let s = Session(edge: edge)
        if dockClick {
            s.pendingDockInteraction = true; s.hasReleasedDockClick = true
            NSEvent.pressedMouseButtons = 1
        } else { advance(1) } // begin the original 220ms outside grace period
        s.rawFrameUnavailable = true
        advance(dockClick ? 1 : 3)
        let keptTracking = s.tracking && s.state == .expanded && s.collapseAnimations == 0 && s.focusReleases == 0
        let hoveredDuringFailure = s.hover()
        s.rawFrameUnavailable = false; NSEvent.pressedMouseButtons = 0
        advance(12)
        let recovered = s.snapped && !s.tracking && !s.isAnimating && s.collapseAnimations == 1 && s.focusReleases == 1
        print("TRACE edge=\(edge) dockClick=\(dockClick) keptTracking=\(keptTracking) hoverDuringFailure=\(hoveredDuringFailure) recovered=\(recovered) state=\(s.state)")
        expect(keptTracking && recovered, "read failure then recovery edge=\(edge) dockClick=\(dockClick)")
        let revealAccepted = s.hover(); advance(4)
        expect(recovered && revealAccepted && s.state == .expanded && s.tracking && s.alpha == 1, "hover works again edge=\(edge) dockClick=\(dockClick)")
    }
    reset(edge: edge)
    let normal = Session(edge: edge)
    advance(2)
    expect(normal.state == .expanded && normal.collapseAnimations == 0, "normal outside grace preserved edge=\(edge)")
    advance(8)
    expect(normal.snapped && !normal.tracking && normal.collapseAnimations == 1 && normal.focusReleases == 1, "normal collapse once edge=\(edge)")
}

for mode in ["pinned", "animating", "fusion", "mouse-down", "sibling", "dock-wait"] {
    reset()
    let s = Session()
    switch mode {
    case "pinned": s.isTemporarilyPinned = true
    case "animating": s.isAnimating = true
    case "fusion": s.fusionHoverLock = true
    case "mouse-down": NSEvent.pressedMouseButtons = 1
    case "sibling": s.siblingUnderMouse = true
    default: s.pendingDockInteraction = true
    }
    advance(10)
    expect(s.tracking && s.state == .expanded && s.collapseAnimations == 0, "existing guard preserved: \(mode)")
}

// Reproduce the reported obscured-window case: the pointer remains inside the
// source window's geometric bounds, but another regular app has taken focus.
reset()
let covered = Session()
NSEvent.mouseLocation = CGPoint(x: 100, y: 700)
NSWorkspace.shared.frontmostApplication = NSRunningApplication(processIdentifier: covered.pid)
advance(2) // populate the stationary-inside cache
NSWorkspace.shared.frontmostApplication = NSRunningApplication(processIdentifier: 456)
advance(10)
expect(covered.snapped && covered.collapseAnimations == 1 && covered.focusReleases == 0,
       "other app focus bypasses cached inside geometry and collapses without restoring focus")
let coveredRevealed = covered.hover(); advance(4)
expect(coveredRevealed && covered.state == .expanded && covered.tracking,
       "hover reveals again after switching to another app")

       "new foreground app ends shortcut wait and ignores background sibling geometry")

for mode in ["pinned", "accessory", "same-app", "brief-switch"] {
    reset()
    let s = Session()
    NSEvent.mouseLocation = CGPoint(x: 100, y: 700)
    let other = NSRunningApplication(processIdentifier: mode == "same-app" ? s.pid : 456)!
    if mode == "accessory" { other.activationPolicy = .accessory }
    if mode == "pinned" { s.isTemporarilyPinned = true }
    NSWorkspace.shared.frontmostApplication = other
    if mode == "brief-switch" {
        advance(2)
        NSWorkspace.shared.frontmostApplication = NSRunningApplication(processIdentifier: s.pid)
    }
    advance(10)
    expect(s.state == .expanded && s.collapseAnimations == 0, "foreground guard preserved: \(mode)")
}

for edge in [1, 2] {
    reset()
    let instant = Session(edge: edge)
    instant.isTemporaryShortcutManaged = true
    expect(instant.directCollapse() && instant.snapped && !instant.isAnimating && instant.collapseAnimations == 0,
           "temporary ordinary collapse is synchronous without animator edge=\(edge)")
    expect(instant.geometryEvents == ["alpha:0.0", "position"] && instant.alpha == 0,
           "temporary collapse hides alpha before moving edge=\(edge)")
    instant.geometryEvents.removeAll()
    expect(instant.hover() && instant.state == .expanded && !instant.isAnimating && instant.tracking && instant.expandAnimations == 0,
           "shared hover entry uses synchronous temporary reveal edge=\(edge)")
    expect(instant.geometryEvents == ["size", "position", "position", "alpha:1.0"] && instant.alpha == 1,
           "temporary reveal resizes and positions before restoring alpha edge=\(edge)")
    _ = instant.directCollapse()
    expect(instant.dockReveal() && instant.pendingDockInteraction && !instant.isAnimating,
           "shared Dock entry is synchronous with interaction gate edge=\(edge)")
    advance(10)
    expect(!instant.isAnimating && instant.expandAnimations == 0 && instant.collapseAnimations == 0,
           "temporary transitions never queue animation callbacks edge=\(edge)")
}

for edge in [1, 2] {
    reset(edge: edge)
    let moved = Session(edge: edge)
    moved.isTemporaryShortcutManaged = true
    moved.pendingDockInteraction = true
    moved.hasReleasedDockClick = true
    moved.outsideCollapseCandidateSince = Date(-1)
    NSEvent.pressedMouseButtons = 1
    advance(10)
    expect(moved.state == .expanded && moved.geometryEvents.isEmpty && moved.outsideCollapseCandidateSince == nil,
           "dragging outside old frame never collapses even during Dock interaction wait edge=\(edge)")
    expect(!moved.directCollapse(), "other collapse entry points reject temporary window during drag edge=\(edge)")

    moved.cgFrameOverride = moved.frame // Simulate CG still reporting the pre-drag edge frame.
    moved.cachedInteractionHitFrame = moved.frame
    moved.cachedInteractionHitFrameAt = Date()
    moved.frame = CGRect(x: 250, y: 180, width: 410, height: 320)
    NSEvent.mouseLocation = CGPoint(x: 450, y: 420)
    advance(3)
    NSEvent.pressedMouseButtons = 0
    advance(10) // Deliberately do not deliver an AX geometry notification.
    expect(moved.state == .expanded && moved.frame == CGRect(x: 250, y: 180, width: 410, height: 320)
           && moved.geometryEvents.isEmpty && !moved.pendingDockInteraction,
           "drop stays put and hit testing uses current AX frame despite stale CG bounds edge=\(edge)")
    expect(moved.measuredFrame()?.width == 410 && moved.lockedWidth == 410,
           "temporary geometry tracks small width changes without stale width lock edge=\(edge)")

    expect(moved.directCollapse() && moved.snapped, "dragged window collapses normally edge=\(edge)")
    moved.revealForTemporaryStashTab()
    expect(moved.frame == CGRect(x: edge == 1 ? 0 : 590, y: 30, width: 410, height: 730),
           "production reveal ignores dragged Y and height, fills usable height and retains width on original edge=\(edge)")

    // A stationary pointer outside a window moved by the app must not stay
    // indefinitely inside the old cached frame when AX notifications are late.
    NSEvent.mouseLocation = CGPoint(x: moved.frame.midX, y: 800 - moved.frame.midY)
    advance(2)
    moved.frame.origin.y = 550
    advance(10)
    expect(moved.snapped, "stationary pointer uses new geometry without AX notification edge=\(edge)")
}

for edge in [1, 2] {
    reset(edge: edge)
    let constrained = Session(edge: edge)
    constrained.isTemporaryShortcutManaged = true
    constrained.maximumHeight = 500
    _ = constrained.directCollapse()
    constrained.revealForTemporaryStashTab()
    expect(constrained.frame == CGRect(x: edge == 1 ? 0 : 600, y: 145, width: 400, height: 500),
           "production reveal centers read-back maximum height in usable area edge=\(edge)")
    constrained.frame = CGRect(x: 250, y: 650, width: 417, height: 240)
    _ = constrained.directCollapse()
    constrained.revealForTemporaryStashTab()
    expect(constrained.frame == CGRect(x: edge == 1 ? 0 : 583, y: 145, width: 417, height: 500),
           "constrained reveal ignores dragged height and Y while retaining new width edge=\(edge)")
    constrained.screen.visibleFrame = CGRect(x: 0, y: 40, width: 1000, height: 300)
    _ = constrained.directCollapse()
    constrained.revealForTemporaryStashTab()
    expect(constrained.frame == CGRect(x: edge == 1 ? 0 : 583, y: 460, width: 417, height: 300),
           "reveal recomputes usable geometry after visible area changes edge=\(edge)")

    reset(edge: edge)
    let configured = Session(edge: edge)
    configured.frame.origin.y = 80
    _ = configured.directCollapse()
    advance(4)
    let configuredRevealed = configured.dockReveal()
    advance(5)
    expect(configuredRevealed && configured.frame.height == 300 && configured.frame.minY == 80,
           "settings-managed window retains original height and Y instead of normalizing to usable area edge=\(edge)")
}
