var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}
func enumerate(_ system: WindowSwitcherSystem) -> [WindowSwitcherInventory] {
    var snapshots: [WindowSwitcherInventory] = []
    system.enumerate { snapshots.append($0) }
    TestDispatchQueue.drain()
    return snapshots
}
