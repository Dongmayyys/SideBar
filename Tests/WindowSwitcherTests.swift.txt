var total = 0
var failures = 0
func expect(_ condition: Bool, _ name: String) {
    total += 1
    if !condition { failures += 1 }
    print("\(condition ? "PASS" : "FAIL") \(name)")
}
func match(_ query: String, _ text: String) -> WindowSwitcherMatch? {
    WindowSwitcherSearch.match(query: query, fields: [WindowSwitcherSearchField(text)])
}
expect(match("code", "Code") == .exact, "exact match is case insensitive")
expect(match("  CODE \n", "Code") == .exact, "query trims whitespace")
expect(match("co", "Code") == .prefix, "prefix rank")
expect(match("studio", "Visual Studio Code") == .wordStart, "word-start rank")
expect(match("studio code", "Visual Studio Code") == .wordStart, "multi-word suffix")
expect(match("vsc", "Visual Studio Code") == .initials, "acronym rank")
expect(match("vscode", "Visual Studio Code") == .subsequence, "non-contiguous rank")
expect(match("soe", "Visual Studio Code") == .subsequence, "ordered subsequence")
expect(match("evs", "Visual Studio Code") == nil, "reversed characters do not match")
expect(match("zzz", "Visual Studio Code") == nil, "non-matches are omitted")
expect(match("", "Code") == .exact, "empty query includes all candidates")
expect(match("cafe", "Café") == .exact, "Latin diacritics are ignored")
expect(match("chrome", "Google Chrome") == .wordStart, "application word-start")
expect(match("gc", "Google Chrome") == .initials, "application acronym")
expect(match("vs", "VisualStudio") == .initials, "camel-case acronym")
expect(match("dd", "滴答清单") == .initials, "dd matches pinyin initials")
expect(match("dida", "滴答清单") == .prefix, "dida matches full-pinyin prefix")

func candidate(_ id: UInt32, _ title: String) -> WindowSwitcherCandidate {
    WindowSwitcherCandidate(id: id, pid: 100, appName: "Editor", title: title)
}
var candidates = [candidate(1, "Code"), candidate(2, "Code notes"), candidate(3, "Visual Code"),
                  candidate(4, "Cool Orange Dog Elephant"), candidate(5, "Chocolate fudge")]
var recency = WindowSwitcherRecency()
recency.seed(stackingOrder: [5, 4, 3, 2, 1])
func sorted(_ query: String, _ values: [WindowSwitcherCandidate] = candidates,
            ownPID: Int32 = 999, focused: UInt32? = nil, stashed: Set<UInt32> = []) -> [UInt32] {
    let fields = Dictionary(uniqueKeysWithValues: values.map { ($0.id, [WindowSwitcherSearchField($0.appName), WindowSwitcherSearchField($0.title)]) })
    return WindowSwitcherSearch.sortedIDs(query: query, candidates: values, fields: fields,
        recency: recency, ownPID: ownPID, focusedID: focused, stashedIDs: stashed)
}
expect(sorted("code") == [1, 2, 3, 4, 5], "all five grades outrank inverse recency")
expect(sorted("") == [5, 4, 3, 2, 1], "initial order is WindowServer stacking order")
recency.recordFocus(2)
expect(sorted("") == [2, 5, 4, 3, 1], "focus notification promotes one exact window")
expect(sorted("code") == [1, 2, 3, 4, 5], "new focus never outranks a better match")
let equal = [candidate(1, "Code"), candidate(2, "Code"), candidate(3, "Code")]
expect(sorted("code", equal) == [2, 3, 1], "same grade sorts by recent focus")
recency.recordFocus(1)
expect(sorted("code", equal) == [1, 2, 3], "same application windows have independent recency")
recency.seed(stackingOrder: [3, 2, 1])
expect(sorted("code", equal) == [1, 2, 3], "later CG snapshots do not overwrite observed focus")
expect(sorted("nothing") == [], "non-matching windows disappear")
expect(sorted("", equal, focused: 2) == [1, 3], "exclude only currently focused window")
expect(sorted("", equal, stashed: [1, 3]) == [2], "exclude live stash IDs")
expect(sorted("", equal, stashed: [99]) == [1, 2, 3], "unrelated or pending IDs do not exclude windows")
expect(sorted("", equal, ownPID: 100).isEmpty, "exclude SideBar's own process")
var minimized = candidate(6, "Minimized")
minimized.isMinimized = true
expect(sorted("", [minimized]) == [6], "minimized window remains switchable")
var background = minimized
background.hasSpace = false
expect(sorted("", [background]) == [6], "AX-confirmed minimized window is restorable even with no reported Space")
background.isMinimized = false
expect(sorted("", [background]).isEmpty, "no-Space non-minimized background windows are excluded")
var utility = candidate(7, "Utility")
utility.isRegularApplication = false
expect(sorted("", [utility]).isEmpty, "non-regular applications are excluded")
utility.isRegularApplication = true
utility.isStandardWindow = false
expect(sorted("", [utility]).isEmpty, "non-standard AX windows are excluded")
utility.isStandardWindow = true
utility.hasValidSize = false
expect(sorted("", [utility]).isEmpty, "invalid window size is excluded")
expect(sorted("min", [minimized], focused: 6).isEmpty, "exclusion applies before matching")
expect(sorted("", [candidate(8, "Other display/Space")]) == [8], "eligible other-Space windows are not filtered by onscreen state")
let ties = [candidate(20, "Tie"), candidate(10, "Tie")]
expect(sorted("tie", ties) == [20, 10], "unobserved ties preserve inventory stacking order")
recency.retain([1, 2, 3])
expect(recency.order[5] == nil && recency.order[1] != nil, "closed WindowServer IDs are pruned")

// Grouping consumes the real matcher and excludes windows before deciding single vs grouped rows.
let groupedCandidates = [
    WindowSwitcherCandidate(id: 101, pid: 11, appName: "Editor", title: "Report"),
    WindowSwitcherCandidate(id: 102, pid: 11, appName: "Editor", title: "Notes"),
    WindowSwitcherCandidate(id: 201, pid: 22, appName: "Browser", title: "Report draft"),
    WindowSwitcherCandidate(id: 301, pid: 33, appName: "Viewer", title: "Other")
]
var groupRecency = WindowSwitcherRecency()
groupRecency.seed(stackingOrder: [102, 201, 301, 101])
func grouped(_ query: String, _ values: [WindowSwitcherCandidate] = groupedCandidates,
             focused: UInt32? = nil, stashed: Set<UInt32> = []) -> [WindowSwitcherRow] {
    let fields = Dictionary(uniqueKeysWithValues: values.map {
        ($0.id, [WindowSwitcherSearchField($0.appName), WindowSwitcherSearchField($0.title)])
    })
    return WindowSwitcherList.rows(query: query, candidates: values, fields: fields, recency: groupRecency,
                                   ownPID: 999, focusedID: focused, stashedIDs: stashed)
}
let groupedEmpty = grouped("")
expect(groupedEmpty.map(\.targetID) == [102, 101, 201, 301], "one row per window; same-software windows stay adjacent in recency order")
expect(groupedEmpty.first?.id == .window(102), "every row is a window selection")
let reportRows = grouped("report")
expect(reportRows.map(\.targetID) == [101, 201], "only matching windows are listed")
expect(reportRows[0].titleHighlights == [NSRange(location: 0, length: 6)], "window rows carry title hit ranges")
expect(reportRows[1].titleHighlights == [NSRange(location: 0, length: 6)], "other software rows carry title hit ranges")
let editorRows = grouped("EDITOR")
expect(editorRows.map(\.targetID) == [102, 101], "application name hit includes all eligible windows")
expect(editorRows.first?.appHighlights == [NSRange(location: 0, length: 6)], "rows carry software-name highlights")
expect(editorRows[0].titleHighlights.isEmpty, "app-only hit does not invent a title highlight")
expect(grouped("missing").isEmpty, "no matching application or title produces no rows")
expect(grouped("", focused: 102).map(\.targetID) == [201, 301, 101], "focused window is excluded")
expect(grouped("", stashed: [101, 102]).map(\.targetID) == [201, 301], "stashed windows are excluded")
let sameNames = [
    WindowSwitcherCandidate(id: 401, pid: 41, appName: "Same", title: "First"),
    WindowSwitcherCandidate(id: 402, pid: 42, appName: "Same", title: "Second")
]
expect(grouped("", sameNames).map(\.targetID) == [401, 402], "same display name lists each window")
var sameBundle = sameNames
sameBundle[0].bundleID = "test.same"
sameBundle[1].bundleID = "test.same"
expect(grouped("", sameBundle).count == 2, "same software bundle lists each window")
let interleaved = [
    WindowSwitcherCandidate(id: 501, pid: 51, appName: "A", title: "Code"),
    WindowSwitcherCandidate(id: 502, pid: 52, appName: "B", title: "Code notes"),
    WindowSwitcherCandidate(id: 503, pid: 51, appName: "A", title: "Visual Code")
]
expect(grouped("code", interleaved).map(\.targetID) == [501, 503, 502], "same-software windows stay together without changing within-group match ranks")

func highlighted(_ query: String, _ source: String) -> [NSRange] {
    WindowSwitcherSearchField(source).match(WindowSwitcherSearchField.normalize(query))?.ranges ?? []
}
expect(highlighted("dd", "滴答清单") == [NSRange(location: 0, length: 2)], "dd highlights 滴答 rather than Latin offsets")
expect(highlighted("dida", "滴答清单") == [NSRange(location: 0, length: 2)], "full-pinyin prefix maps every syllable back to its Hanzi")
expect(highlighted("qd", "滴答清单") == [NSRange(location: 2, length: 2)], "interior pinyin initials highlight 清单")
expect(highlighted("qingdan", "滴答清单") == [NSRange(location: 2, length: 2)], "word-start pinyin maps to source Hanzi")
expect(highlighted("didaqingdan", "滴答清单") == [NSRange(location: 0, length: 4)], "exact full-pinyin hit highlights all Hanzi once")
expect(highlighted("清单", "滴答清单") == [NSRange(location: 2, length: 2)], "direct Chinese hit uses original UTF-16 offsets")
expect(highlighted("vsc", "Visual Studio Code") == [NSRange(location: 0, length: 1), NSRange(location: 7, length: 1), NSRange(location: 14, length: 1)], "English acronym highlights word initials")
expect(highlighted("soe", "Visual Studio Code") == [NSRange(location: 2, length: 1), NSRange(location: 12, length: 1), NSRange(location: 17, length: 1)], "subsequence highlights only the ordered matching characters")
expect(highlighted("studio code", "Visual Studio Code") == [NSRange(location: 7, length: 11)], "multi-word match includes correct source spacing")
expect(highlighted("cafe", "Café") == [NSRange(location: 0, length: 4)], "accent-folded exact match keeps original characters")
expect(highlighted("ss", "Straße") == [("Straße" as NSString).range(of: "ß")], "case-fold expansion maps two matched letters to one source character")
let emojiTitle = "👩🏽‍💻 滴答清单.md"
expect(highlighted("qd", emojiTitle) == [(emojiTitle as NSString).range(of: "清单")], "emoji UTF-16 width does not displace pinyin highlights")
expect(highlighted("chong", "重庆计划") == [NSRange(location: 0, length: 1)], "whole-string polyphonic transliteration maps chong to 重")
expect(highlighted("cq", "重庆计划") == [NSRange(location: 0, length: 2)], "contextual pinyin initials map to 重庆")
expect(highlighted("", "滴答清单").isEmpty, "empty query does not bold all results")
expect(highlighted("zzz", "滴答清单").isEmpty, "non-hit text has no bold spans")
expect(highlighted("qd", "  滴答清单  ") == [NSRange(location: 4, length: 2)], "trimming does not shift source ranges")
let combining = "Cafe\u{301}"
expect(highlighted("e", combining) == [(combining as NSString).range(of: "e\u{301}")], "decomposed accent remains a complete composed-character range")

var selection = WindowSwitcherSelection()
let selectionIDs = groupedEmpty.map(\.id)
selection.reconcile(selectionIDs, preservingSelection: false)
expect(selection.selectedID == selectionIDs[0], "default selection is first row, including headers")
selection.hover(selectionIDs[2])
expect(selection.selectedID == selectionIDs[2], "hover selects the row under the mouse")
selection.move(to: selectionIDs[3])
expect(selection.selectedID == selectionIDs[3], "keyboard takes precedence over old hover selection")
selection.reconcile(selectionIDs, preservingSelection: true)
expect(selection.selectedID == selectionIDs[3], "passive refresh cannot reapply stationary mouse hover")
selection.hover(nil)
expect(selection.selectedID == selectionIDs[3], "mouse exit preserves latest keyboard selection")
selection.hover(selectionIDs[1])
expect(selection.selectedID == selectionIDs[1], "fresh mouse movement can select a header child again")
selection.hover(nil)
expect(selection.selectedID == selectionIDs[3], "leaving hovered row restores keyboard selection")
selection.reconcile([], preservingSelection: true)
expect(selection.selectedID == nil, "empty results clear selection")

let screen = CGRect(x: -1440, y: 100, width: 1440, height: 900)
let smallPanel = WindowSwitcherPanelLayout.frame(visibleFrame: screen, rowCount: 1)
let emptyPanel = WindowSwitcherPanelLayout.frame(visibleFrame: screen, rowCount: 0)

expect(smallPanel.midY > screen.midY, "short panel is in upper-middle screen area")
expect(smallPanel.maxY == largePanel.maxY, "height changes preserve the top anchor")
expect(largePanel.height == 480 && largePanel.height > smallPanel.height, "panel grows with rows up to scroll height cap")
expect(emptyPanel == smallPanel, "empty results retain a compact readable status area")
expect(screen.contains(largePanel), "panel stays within a negative-origin secondary display")
let shortScreen = CGRect(x: 0, y: 0, width: 500, height: 300)
expect(shortScreen.contains(WindowSwitcherPanelLayout.frame(visibleFrame: shortScreen, rowCount: 50)), "height and width limits adapt to small displays")

let actions = SwitcherActionHarness()
actions.rows = groupedEmpty
actions.inventory.candidates = groupedCandidates
actions.click(0)
expect(actions.system.focused == [102] && actions.closes == 1, "one click closes and focuses that window")
actions.click(1)
expect(actions.system.focused == [102, 101] && actions.closes == 2, "one click on a child confirms that exact window without double-click")
actions.click(-1)
expect(actions.system.focused.count == 2 && actions.closes == 2, "clicking blank table space cannot confirm a previous selection")
actions.excluded = [201]
actions.click(2)
expect(actions.system.focused.count == 2 && actions.closes == 3, "new stash exclusion is rechecked at mouse confirmation")
actions.inventory.candidates.removeAll { $0.id == 301 }
actions.enter(3)
expect(actions.system.focused.count == 2 && actions.closes == 4, "missing keyboard target closes the panel without focusing another window")
expect(actions.system.failures.count == 2, "excluded or missing selected target emits failure feedback")
actions.enter(-1)
expect(actions.system.failures.count == 2, "confirming an empty selection does not emit a false error")
let footerPanel = WindowSwitcherPanelLayout.frame(visibleFrame: screen, rowCount: 1, statusFooter: true)
expect(footerPanel.height == smallPanel.height + 24 && footerPanel.maxY == smallPanel.maxY,
       "partial inventory status has space below list without moving the top anchor")
print("RESULT window-switcher checks=\(total) failures=\(failures)")
exit(failures == 0 ? 0 : 1)
